# utils/universe.py
import pandas as pd

def eligible_mask_per_date(prices, volume, min_price, adv_window, min_adv_usd):
    adv_usd = (prices * volume).rolling(adv_window).mean()
    mask = (prices >= min_price) & (adv_usd >= min_adv_usd)
    return mask.astype(bool)

def birth_death_mask(prices, min_days_from_ipo=60):
    first_seen = prices.apply(lambda s: s.first_valid_index())
    birth_ok = pd.DataFrame(index=prices.index, columns=prices.columns, data=False)
    for c in prices.columns:
        if first_seen[c] is None: 
            continue
        birth_ok.loc[prices.index >= (first_seen[c] + pd.tseries.offsets.BDay(min_days_from_ipo)), c] = True
    # death = brak dalszych notowań → maska False po ostatniej dacie notowanej
    last_seen = prices.apply(lambda s: s.last_valid_index())
    for c in prices.columns:
        if last_seen[c] is None: 
            continue
        birth_ok.loc[prices.index > last_seen[c], c] = False
    return birth_ok

# utils/exec_policy.py
import pandas as pd
def plan_twap(delta_w, prices, adv_usd, spread, equity, window_minutes=60, slices=6, adv_cap=0.1, gamma=0.25):
    """Zwraca DataFrame z 'slice_time, symbol, side, qty_usd, limit_price'."""
    # docelowa $-kwota do przesunięcia na symbol
    dollars = (delta_w.abs() * equity * prices)
    # cap per dzień
    cap_dollars = adv_usd * adv_cap
    dollars = dollars.clip(upper=cap_dollars)
    rows = []
    for sym in dollars.columns:
        total = float(dollars.iloc[-1][sym])  # przyjmij dzisiejszy rebalans
        if total <= 0: 
            continue
        per_slice = total / slices
        side = "BUY" if delta_w.iloc[-1][sym] > 0 else "SELL"
        mid = float(prices.iloc[-1][sym])
        lim = mid * (1 + (gamma*float(spread.iloc[-1][sym])) * (1 if side=="BUY" else -1))
        for i in range(slices):
            t = pd.Timestamp.utcnow().floor("min") + pd.Timedelta(minutes=i*window_minutes//slices)
            rows.append([t, sym, side, per_slice, lim])
    return pd.DataFrame(rows, columns=["slice_time","symbol","side","qty_usd","limit_price"])

# utils/risk.py
import pandas as pd, numpy as np
def ewma_vol(series, halflife=20):
    return series.ewm(halflife=halflife).std()

def scale_to_target(port_ret, target_vol=0.10, freq="D", halflife=20, min_leverage=0.5, max_leverage=2.0):
    ann = {"D":252,"W":52,"M":12}[freq.upper()]
    rv = ewma_vol(port_ret, halflife=halflife) * np.sqrt(ann)
    lev = (target_vol / rv).clip(lower=min_leverage, upper=max_leverage).shift(1).fillna(1.0)
    return lev

3.2 Drawdown clamp (linearnie redukuje gross przy DD)
# utils/risk.py
def dd_clamp(equity_curve, dd_soft=0.08, dd_hard=0.12):
    peak = equity_curve.cummax()
    dd = equity_curve/peak - 1.0
    k = (dd - (-dd_soft)) / (dd_hard - dd_soft)
    clamp = (1 - k).clip(lower=0.0, upper=1.0)
    clamp[dd > -dd_soft] = 1.0
    return clamp

# w report_daily.py:

    return hashlib.sha1(text.encode()).hexdigest()

def post_message(channel: str, text: str, retries: int=3):
    client = _get_client()
    key = _idempotency_key(text)
    for i in range(retries):
        try:
            client.chat_postMessage(channel=channel, text=text, username="alpha-lab bot", metadata={"event_type":"alpha_report","event_payload":{"key":key}})
            return
        except SlackApiError as e:
            if i == retries-1: raise

import json, hashlib, time
def cfg_hash(cfg: dict) -> str:
    return hashlib.sha1(json.dumps(cfg, sort_keys=True).encode()).hexdigest()
def append_ledger(path, cfg_hash):
    os.makedirs(os.path.dirname(path), exist_ok=True)
    L = []
    if os.path.exists(path): L = json.load(open(path))
    L.append({"ts": time.time(), "hash": cfg_hash})
    json.dump(L, open(path,"w"))
    return len(L)
