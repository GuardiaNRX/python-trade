+from dataclasses import dataclass

+@dataclass
+class ICRegimeCfg:
+    low_enter: float = -0.05   # IC3m - IC12m <= enter → start low-streak
+    low_exit:  float = -0.02   # wyjście z low po przejściu powyżej
+    high_enter: float = 0.03   # IC3m - IC12m >= enter → start high-streak
+    high_exit:  float = 0.01   # wyjście z high po spadku poniżej
+    low_mult: float = 0.50
+    base_mult: float = 1.00
+    high_mult: float = 1.25
+    min_streak: int = 10       # min liczba kolejnych sesji do wejścia w reżim
+
+def _load_state(path: str) -> dict:
+    if os.path.exists(path):
+        try: return json.load(open(path))
+        except Exception: pass
+    return {"regime":"base","low_streak":0,"high_streak":0}
+
+def _save_state(path: str, state: dict):
+    os.makedirs(os.path.dirname(path), exist_ok=True)
+    with open(path,"w") as f: json.dump(state,f)
+
+def decide_multiplier(ic3: float|None, ic12: float|None, cfg: ICRegimeCfg, state_path="backtests/results/ic_state.json"):
+    """Z histerezą i streakiem; zwraca (mult, new_state)."""
+    st = _load_state(state_path)
+    if ic3 is None or ic12 is None:
+        return cfg.base_mult, st
+    gap = ic3 - ic12
+    # update streaks
+    st["low_streak"]  = st.get("low_streak",0)  + (1 if gap <= cfg.low_enter else 0) if gap <= cfg.low_enter else 0
+    st["high_streak"] = st.get("high_streak",0) + (1 if gap >= cfg.high_enter else 0) if gap >= cfg.high_enter else 0
+    regime = st.get("regime","base")
+    # enter conditions
+    if regime != "low"  and gap <= cfg.low_enter  and st["low_streak"]  >= cfg.min_streak: regime = "low"
+    if regime != "high" and gap >= cfg.high_enter and st["high_streak"] >= cfg.min_streak: regime = "high"
+    # exit conditions (hysteresis)
+    if regime == "low"  and gap >= cfg.low_exit:  regime = "base"
+    if regime == "high" and gap <= cfg.high_exit: regime = "base"
+    st["regime"] = regime
+    mult = {"low":cfg.low_mult, "base":cfg.base_mult, "high":cfg.high_mult}[regime]
+    _save_state(state_path, st)
+    return mult, st

+def _key(prefix: str, snapshot_id: str, **params) -> str:
+    s = json.dumps({"snapshot":snapshot_id,"params":params}, sort_keys=True)

+def append_slo(history_path: str, date_str: str, step_times: dict):
+    row = {"date": date_str} | {f"t_{k}": float(v) for k,v in step_times.items()}

+def rolling_warn(history_path: str, col: str, window:int=5, budget: float|None=None, factor: float=1.2):
+    if not os.path.exists(history_path): return None
+    df = pd.read_csv(history_path)
+    if col not in df.columns or len(df) < window: return None
+    r = df[col].tail(window).mean()
+    thr = (budget * factor) if budget else None
+    if thr and r > thr:
+        return f"{col} 5d avg {r:.0f}s > budget {thr:.0f}s"
+    return None
