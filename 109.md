+import os, time, argparse, glob, shutil
+
+def _prune_by_days(path, days):
+    if not os.path.exists(path): return
+    cutoff = time.time() - days*86400
+    for p in glob.glob(os.path.join(path, "*")):
+        try:
+            if os.path.isdir(p):
+                if max(os.path.getmtime(r) for r,_,_ in os.walk(p)) < cutoff:
+                    shutil.rmtree(p, ignore_errors=True)
+            else:
+                if os.path.getmtime(p) < cutoff:
+                    os.remove(p)
+        except Exception:
+            pass
+
+def main():
+    ap = argparse.ArgumentParser()
+    ap.add_argument("--reports_days", type=int, default=30)
+    ap.add_argument("--orders_days", type=int, default=90)
+    ap.add_argument("--snapshots_days", type=int, default=365)
+    args = ap.parse_args()
+    _prune_by_days("backtests/reports", args.reports_days)
+    _prune_by_days("backtests/results/orders", args.orders_days)
+    _prune_by_days("data/source_snapshots", args.snapshots_days)
+
+if __name__ == "__main__":
+    main()

+import pandas as pd
+def universe_drift(today_mask: 'pd.Series|pd.DataFrame', yday_mask: 'pd.Series|pd.DataFrame'):
+    t = today_mask.astype(bool)
+    y = yday_mask.astype(bool)
+    if hasattr(t, "to_frame"): t = t.iloc[0] if t.ndim>1 else t
+    if hasattr(y, "to_frame"): y = y.iloc[0] if y.ndim>1 else y
+    added = t.index[t & ~y].tolist()
+    removed = t.index[~t & y].tolist()
+    return {"n_added": len(added), "n_removed": len(removed), "added": added, "removed": removed}

+import os, time, math
+import pandas as pd
+
+def _cache_dir():
+    d = "data/http_cache"
+    os.makedirs(d, exist_ok=True)
+    return d
+
+def _sym_path(sym: str): return os.path.join(_cache_dir(), f"{sym}.parquet")
+
+def _fetch_yf(sym: str, start: str|None, end: str|None, timeout: int=15):
+    import yfinance as yf
+    # yfinance nie ma bezpośredniego timeoutu; rely on underlying requests default
+    return yf.Ticker(sym).history(start=start, end=end, auto_adjust=False)
+
+def get_ohlcv_yf(symbols: list[str], start: str|None, end: str|None,
+                 retries:int=2, backoff: float=1.5) -> tuple[pd.DataFrame,pd.DataFrame,pd.DataFrame,pd.DataFrame]:
+    px, vol, hi, lo = {}, {}, {}, {}
+    for s in symbols:
+        df = None
+        pth = _sym_path(s)
+        if os.path.exists(pth):
+            try:
+                c = pd.read_parquet(pth)
+                df = c
+            except Exception:
+                df = None
+        if df is None or (end and (not c.index.empty and str(c.index.max().date()) < str(pd.to_datetime(end).date()))):
+            # fetch z retry
+            last_err = None
+            for t in range(retries+1):
+                try:
+                    f = _fetch_yf(s, start=start, end=end)
+                    if f is not None and not f.empty:
+                        f.to_parquet(pth)
+                        df = f
+                        break
+                except Exception as e:
+                    last_err = e
+                    time.sleep(backoff**t)
+            if df is None:
+                raise RuntimeError(f"yfinance fetch failed for {s}: {last_err}")
+        if df is None or df.empty:
+            continue
+        # Oczekujemy kolumn: 'Adj Close','Volume','High','Low'
+        if "Adj Close" in df: px[s] = df["Adj Close"].rename(s)
+        elif "Close" in df:   px[s] = df["Close"].rename(s)
+        if "Volume" in df:    vol[s] = df["Volume"].rename(s)
+        if "High" in df:      hi[s]  = df["High"].rename(s)
+        if "Low" in df:       lo[s]  = df["Low"].rename(s)
+    # sklej
+    px  = pd.concat(px.values(), axis=1) if px else pd.DataFrame()
+    vol = pd.concat(vol.values(), axis=1) if vol else pd.DataFrame()
+    hi  = pd.concat(hi.values(), axis=1) if hi else pd.DataFrame()
+    lo  = pd.concat(lo.values(), axis=1) if lo else pd.DataFrame()
+    return px, vol, hi, lo
