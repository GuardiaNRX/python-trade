3) Freshness multi-venue (USA/EU/APAC)
Dlaczego tak

Jedna bramka świeżości dla XNYS nie wystarczy przy wszechświecie wielorynkowym. Chcemy osobne progi pokrycia per region, na bazie faktycznych kalendarzy rynków, oraz jasny raport „kto odstaje”.

Co robimy

Mapujemy symbol → exchange → region.

Liczymy oczekiwaną ostatnią sesję per exchange.

Sprawdzamy pokrycie danych (nie-NaN) na tej dacie per region i porównujemy z progami (np. USA 95%, EU 90%, APAC 90%).

Fail-closed: jeśli któryś region nie spełnia progu — przerywamy run i wysyłamy Slack z listą kilku symboli do diagnostyki.

Wstawka (nowy moduł)
# utils/freshness_multi.py
import pandas as pd, pandas_market_calendars as mcal

def expected_last_session_for_exchanges(exchanges: list[str], asof=None) -> dict[str, pd.Timestamp]:
    asof = pd.Timestamp.utcnow().normalize() if asof is None else pd.Timestamp(asof)
    out = {}
    for ex in exchanges:
        cal = mcal.get_calendar(ex)
        sched = cal.schedule(start_date=asof - pd.Timedelta(days=10), end_date=asof + pd.Timedelta(days=1))
        out[ex] = pd.Timestamp(sched.index.max()).tz_localize(None)
    return out

def region_freshness_gate(prices: pd.DataFrame, mapping: pd.DataFrame, thresholds: dict[str, float]) -> tuple[dict,str|None]:
    """
    mapping: DataFrame[symbol, exchange, region]; thresholds: {"USA":0.95,"EU":0.90,"APAC":0.90}
    Zwraca: {"USA":0.97,...}, failing_region (lub None)
    """
    ex_last = expected_last_session_for_exchanges(sorted(mapping["exchange"].unique().tolist()))
    cov_by_region = {}
    failing = None
    for region, thr in thresholds.items():
        subs = mapping[mapping["region"] == region]
        if subs.empty:
            cov_by_region[region] = 1.0
            continue
        # coverage: symbol ma nie-NaN na swojej dacie exchange
        ok = 0
        for _, row in subs.iterrows():
            sym, ex = row["symbol"], row["exchange"]
            dt = ex_last.get(ex)
            if dt is None or dt not in prices.index: 
                continue
            val = prices.at[dt, sym] if sym in prices.columns else None
            ok += int(pd.notna(val))
        cov = ok / max(1, len(subs))
        cov_by_region[region] = cov
        if failing is None and cov < thr:
            failing = region
    return cov_by_region, failing

4) Sanity HL dla spreadu (Corwin–Schultz robust)
Dlaczego tak

CS na czystym HL bywa „toksyczny” przy splitach/gapach/zerowym wolumenie. Chcemy zabezpieczyć spread:

winsoryzować log-range,

wycinać dni „niewiarygodne” (|log-range| > X·σ i brak wolumenu),

w tych dniach fallback: median-spread z okna / przekrojowa median / stała.

Wstawka (rozszerzony spread)
# utils/spread_robust.py
import numpy as np, pandas as pd
from .impact import corwin_schultz_spread

def robust_cs_spread(high: pd.DataFrame, low: pd.DataFrame, volume: pd.DataFrame,
                     window:int=2, sanity_sigma:float=5.0, median_window:int=20, fallback_bps:float=25.0) -> pd.DataFrame:
    # wstępny CS
    cs = corwin_schultz_spread(high, low, window=window)
    # log-range sanity
    logrng = (np.log(high) - np.log(low)).abs()
    rng_mu = logrng.rolling(median_window, min_periods=5).median()
    rng_sd = logrng.rolling(median_window, min_periods=5).std()
    vol_zero = (volume <= 0) | volume.isna()
    outlier = (logrng > (rng_mu + sanity_sigma * rng_sd)) & vol_zero
    cs = cs.mask(outlier)  # NaN zamiast podejrzanych wycen
    # fallback 1: rolling median
    cs_med = cs.rolling(median_window, min_periods=5).median()
    cs = cs.fillna(cs_med)
    # fallback 2: cross-sectional median w danym dniu
    cs = cs.apply(lambda row: row.fillna(row.median()), axis=1)
    # fallback 3: stała
    cs = cs.fillna(fallback_bps/1e4)
    # clamp rozsądny zakres (0–5%)
    return cs.clip(lower=0.0, upper=0.05)

5) Secrets: menedżer tajemnic + walidacja + fail-closed
Dlaczego tak

ENV są wygodne, ale nietrwałe i podatne na wycieki. Chcemy:

konfigurowalny backend (ENV / AWS SM / GCP SM / Vault),

walidację kluczy krytycznych na starcie (Slack token, kanał — inaczej stop),

Wstawka (uniwersalny loader)
# utils/secrets.py
import os, json
from dataclasses import dataclass

@dataclass
class SecretSpec:
    name: str
    required: bool = True

class Secrets:
    def __init__(self, backend: str|None=None):
        self.backend = (backend or os.environ.get("SECRETS_BACKEND","env")).lower()

    def get(self, key: str) -> str|None:
        if self.backend == "env":
            return os.environ.get(key)
        if self.backend == "aws":
            import boto3
            arn = os.environ.get("AWS_SECRET_ARN")
            if not arn: return None
            val = boto3.client("secretsmanager").get_secret_value(SecretId=arn)["SecretString"]
            return json.loads(val).get(key)
        if self.backend == "gcp":
            from google.cloud import secretmanager
            name = os.environ.get("GCP_SECRET_NAME")
            client = secretmanager.SecretManagerServiceClient()
            resp = client.access_secret_version(request={"name": name})
            val = resp.payload.data.decode("UTF-8")
            return json.loads(val).get(key)
        if self.backend == "vault":
            import hvac
            url = os.environ.get("VAULT_ADDR"); token = os.environ.get("VAULT_TOKEN")
            path = os.environ.get("VAULT_PATH", "secret/data/alpha")
            cli = hvac.Client(url=url, token=token)
            val = cli.secrets.kv.v2.read_secret_version(path=path)["data"]["data"]
            return val.get(key)
        return None

    def validate(self, specs: list[SecretSpec]) -> list[str]:
        missing = []
        for spec in specs:
            if spec.required and not self.get(spec.name):
                missing.append(spec.name)
        return missing

6) Testy integracyjne: golden snapshot + odtwarzalność
Dlaczego tak

Każda zmiana w kodzie powinna być weryfikowana na „złotym” snapshotcie: te same wejścia → te same wyjścia (w granicach tolerancji numerycznej). To zamyka drogę cichym regresjom.

Co robimy

Utrzymujemy katalog ze złotym snapshotem (np. tests/golden/<snapshot_id>/), zawierającym: px.parquet, vol.parquet, hi.parquet, lo.parquet, plus oczekiwane artefakty (report.md, equity.png hash, itp.).

Test integracyjny:

ładuje snapshot jako dane (ominie pobieranie),

uruchamia pipeline w trybie dry-run (bez Slack/Broker),

porównuje hash plików (albo liczby kluczowe z raportu) z oczekiwaniami,

powtarza run 2× i sprawdza identyczność wyników.

Tolerancja: w raporcie ignorujemy linie zawierające Run ID, Data generacji itp.

Wstawka (test)
# tests_integration/test_golden_snapshot.py
import os, hashlib, subprocess, json, re

GOLD = "tests/golden/20250105_abc12345"  # przykład snapshotu
OUT  = "backtests/reports"

def _md5(path):
    with open(path, "rb") as f: return hashlib.md5(f.read()).hexdigest()

def _normalize_md(text: str) -> str:
    # usuń linie z volatile polami
    return "\n".join([ln for ln in text.splitlines() if not re.search(r"(Run ID|Data generacji|Snapshot)", ln)])

def test_golden_replay():
    # 1) uruchom report w trybie dry-run (przygotuj w kodzie obsługę ENV DRY_RUN=1)
    env = dict(os.environ); env["DRY_RUN"]="1"
    ret = subprocess.run(["python","backtests/report_daily.py","--config","configs/example_mom12_1.yaml"], env=env)
    assert ret.returncode==0

    # 2) znajdź najnowszy report.md i policz hash po normalizacji
    md = sorted([p for p in os.listdir(OUT) if p.endswith("_report.md")])[-1]
    text = open(os.path.join(OUT, md), "r", encoding="utf-8").read()
    h1 = hashlib.md5(_normalize_md(text).encode()).hexdigest()

    # 3) porównaj z goldenem
    gold_md = os.path.join(GOLD, "report.md")
    gold_text = open(gold_md,"r",encoding="utf-8").read()
    h_gold = hashlib.md5(_normalize_md(gold_text).encode()).hexdigest()
    assert h1 == h_gold, "Markdown report drifted vs golden snapshot"

    # 4) drugi run (odtwarzalność)
    ret2 = subprocess.run(["python","backtests/report_daily.py","--config","configs/example_mom12_1.yaml"], env=env)
    assert ret2.returncode==0
    md2 = sorted([p for p in os.listdir(OUT) if p.endswith("_report.md")])[-1]
    text2 = open(os.path.join(OUT, md2), "r", encoding="utf-8").read()
    h2 = hashlib.md5(_normalize_md(text2).encode()).hexdigest()
    assert h1 == h2, "Second run differs — pipeline not deterministic"
