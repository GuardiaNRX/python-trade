+import time
+class RuntimeBudget:
+    def __init__(self, max_total_sec=1200, degrade_threshold=0.9):
+        self.t0 = time.time()
+        self.max_total = max_total_sec
+        self.degrade_threshold = degrade_threshold
+        self.steps = {}
+    def start(self, name): self.steps[name] = time.time()
+    def end(self, name): self.steps[name] = time.time() - self.steps.get(name, time.time())
+    @property
+    def elapsed(self): return time.time()-self.t0
+    def should_degrade(self): return self.elapsed >= self.max_total*self.degrade_threshold
+    def left(self): return max(0.0, self.max_total - self.elapsed)
