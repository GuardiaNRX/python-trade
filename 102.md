+import os, time, json, errno
+from contextlib import contextmanager
+
+DEFAULT_LOCK = "backtests/results/job.lock"
+
+def _write_lock(path, data):
+    os.makedirs(os.path.dirname(path), exist_ok=True)
+    with open(path, "w") as f:
+        json.dump(data, f)
+
+def _read_lock(path):
+    try:
+        with open(path, "r") as f:
+            return json.load(f)
+    except Exception:
+        return None
+
+@contextmanager
+def run_lock(path: str = DEFAULT_LOCK, stale_after_sec: int = 3600):
+    """
+    Tworzy lockfile z PID i timestampem. Jeśli istnieje i nie jest „stary” → podnosi błąd.
+    Lock usuwany automatycznie po wyjściu z kontekstu.
+    """
+    now = time.time()
+    if os.path.exists(path):
+        meta = _read_lock(path) or {}
+        if (now - float(meta.get("ts", 0))) < stale_after_sec:
+            raise RuntimeError(f"Another run in progress (lock: {path}; meta={meta})")
+    _write_lock(path, {"ts": now, "pid": os.getpid()})
+    try:
+        yield
+    finally:
+        try:
+            os.remove(path)
+        except OSError as e:
+            if e.errno != errno.ENOENT:
+                raise

*** Add File: utils/risk_limits.py
+import numpy as np
+import pandas as pd
+
+def enforce_gross_limit(weights: pd.DataFrame, max_gross: float=1.5) -> pd.DataFrame:
+    gross = weights.abs().sum(axis=1)
+    scale = (max_gross / gross).clip(upper=1.0).fillna(1.0)
+    return (weights.T * scale).T
+
+def enforce_single_name_limit(weights: pd.DataFrame, max_name: float=0.10) -> pd.DataFrame:
+    return weights.clip(upper=max_name, lower=-max_name)
+
+def turnover_series(weights: pd.DataFrame) -> pd.Series:
+    return weights.diff().abs().sum(axis=1).fillna(0.0)
+
+def limit_turnover(weights: pd.DataFrame, max_turnover: float=0.30) -> pd.DataFrame:
+    to = turnover_series(weights)
+    scale = (max_turnover / to).clip(upper=1.0).replace([np.inf, -np.inf], 1.0).fillna(1.0)
+    adj = (weights.diff().T * scale).T.fillna(0.0)
+    w0 = weights.shift(1).fillna(0.0)
+    return (w0 + adj).clip(-1.0, 1.0)
+
+def kill_switch_by_drawdown(port_ret: pd.Series, dd_hard: float=0.15) -> bool:
+    eq = (1 + port_ret.fillna(0)).cumprod()
+    dd = (eq/eq.cummax() - 1.0).min()
+    return (dd <= -abs(dd_hard))
*** End Patch

+import os, time
+import numpy as np
+import pandas as pd
+from dataclasses import dataclass
+
+@dataclass
+class SLO:
+    step_times: dict
+    n_assets_after_filter: int
+    pct_missing_today: float
+    ic_3m: float
+    ic_12m: float
+    adv_viol_share: float
+
+def alert_thresholds(slo: SLO) -> list[str]:
+    alerts = []
+    if slo.step_times.get("total", 0) > 1200:  # >20 min
+        alerts.append(f"Czas total > 20m ({slo.step_times['total']:.0f}s)")
+    if slo.n_assets_after_filter < 3:
+        alerts.append("Bardzo mało aktywów po filtrze (<3)")
+    if slo.pct_missing_today > 0.1:
+        alerts.append(f"Braki danych dziś >10% ({slo.pct_missing_today:.1%})")
+    if np.isfinite(slo.ic_3m) and np.isfinite(slo.ic_12m) and (slo.ic_3m < slo.ic_12m - 0.05):
+        alerts.append(f"IC 3m {slo.ic_3m:.3f} << 12m {slo.ic_12m:.3f}")
+    if slo.adv_viol_share > 0.05:
+        alerts.append(f"Naruszenia %ADV >5% ({slo.adv_viol_share:.1%})")
+    return alerts
