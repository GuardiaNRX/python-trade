+import os, json

+def decide_backfill(compare_result: dict, policy: str="fail"):
+    """
+    policy: 'fail' | 'flag' | 'allow'
+    """
+    ch = compare_result.get("changed", False)
+    if not ch: 
+        return {"action":"ok"}
+    if policy == "fail":
+        return {"action":"fail", "msg":"Backfill detected: hashes differ."}
+    if policy == "flag":
+        return {"action":"flag", "msg":"Backfill detected (flagged). Continue with caution."}
+    return {"action":"allow"}

*** Add File: utils/broker.py
+import os, time, json, uuid
+import pandas as pd
+from dataclasses import dataclass
+
+@dataclass
+class SubmitResult:
+    ok: bool
+    submitted: int
+    errors: list
+
+class PaperBroker:
+    def __init__(self, out_dir="backtests/results/paper", max_parallel=20, timeout_sec=5, retries=2):
+        os.makedirs(out_dir, exist_ok=True)
+        self.out_dir = out_dir
+        self.max_parallel = max_parallel
+        self.timeout_sec = timeout_sec
+        self.retries = retries
+
+    def positions(self) -> pd.Series:
+        p = os.path.join(self.out_dir, "positions.csv")
+        if not os.path.exists(p): return pd.Series(dtype=float)
+        df = pd.read_csv(p)
+        return df.set_index("symbol")["quantity"].astype(float)
+
+    def submit(self, plan: pd.DataFrame, run_id: str) -> SubmitResult:
+        # throttle
+        plan = plan.copy().head(self.max_parallel)
+        plan["client_id"] = [f"{run_id}_{i}" for i in range(len(plan))]
+        # symulacja filli = wpis do CSV „fills”
+        fills = []
+        errs = []
+        for _, row in plan.iterrows():
+            ok = False
+            for t in range(self.retries+1):
+                try:
+                    time.sleep(0.01)
+                    fills.append({
+                        "ts": pd.Timestamp.utcnow().isoformat(),
+                        "symbol": row["symbol"],
+                        "side": row["side"],
+                        "exec_price": row.get("limit_price", 0.0),
+                        "mid_price": row.get("limit_price", 0.0),
+                        "notional": row.get("qty_usd", 0.0),
+                        "adv_usd": 0.0,
+                        "client_id": row["client_id"]
+                    })
+                    ok = True; break
+                except Exception as e:
+                    if t == self.retries: errs.append(str(e))
+                    time.sleep(2**t)
+            if not ok: errs.append(f"submit_failed:{row['symbol']}")
+        # zapis fills
+        fp = os.path.join(self.out_dir, "fills.csv")
+        pd.DataFrame(fills).to_csv(fp, mode="a", index=False, header=not os.path.exists(fp))
+        return SubmitResult(ok=(len(errs)==0), submitted=len(fills), errors=errs)
+
+    def cancel_stale(self) -> bool:
+        # w paper to tylko log; w realnym brokerze — API cancel open orders
+        p = os.path.join(self.out_dir, "cancels.log")
+        with open(p,"a") as f: f.write(pd.Timestamp.utcnow().isoformat()+" CANCEL ALL\n")
+        return True
*** End Patch
